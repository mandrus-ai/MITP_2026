245void AuthWindow::onAuth()
246{
247 QString login = authLogin->text().trimmed();
248 QString pass = authPass->text().trimmed();
249
250 if(login.isEmpty() || pass.isEmpty()) {
251 authStatus->setText("Please fill in all fields!");
252 authStatus->setStyleSheet("color: red; font-size: 11px;");
253 return;
254 }
255
256 authStatus->setText("Checking...");
257 authStatus->setStyleSheet("color: orange; font-size: 11px;");
258 authBtn->setEnabled(false);
259
260 sendToServer(QString("auth&%1&%2").arg(login).arg(pass));
261}

263void AuthWindow::onReg()
264{
265 QString displayName = regDisplayName->text().trimmed();
266 QString login = regLogin->text().trimmed();
267 QString pass = regPass->text().trimmed();
268 QString confirmPass = regConfirmPass->text().trimmed();
269
270 if(displayName.isEmpty() || login.isEmpty() || pass.isEmpty() || confirmPass.isEmpty()) {
271 regStatus->setText("Please fill in all fields!");
272 regStatus->setStyleSheet("color: red; font-size: 11px;");
273 return;
274 }
275
276 if(pass != confirmPass) {
277 regStatus->setText("Passwords do not match!");
278 regStatus->setStyleSheet("color: red; font-size: 11px;");
279 return;
280 }
281
282 regStatus->setText("Registering...");
283 regStatus->setStyleSheet("color: orange; font-size: 11px;");
284 regBtn->setEnabled(false);
285
286 sendToServer(
287 QString("reg&%1&%2&%3")
288 .arg(login)
289 .arg(pass)
290 .arg(displayName)
291 );
292}

294void AuthWindow::onReadyRead(QString msg)
295{
296 msg.remove(QChar(1));
297 msg = msg.trimmed();
298 qDebug() << "Received:" << msg;
299
300 if(msg.startsWith("auth_success"))
301 {
302 // Формат: auth_success&login|userId|score|coins|level|skin|background|xp
303 QStringList parts = msg.split('&');
304 if(parts.size() >= 2) {
305 QStringList dataParts = parts[1].split('|');
306 if(dataParts.size() >= 8) {
307 QString login = dataParts[0];
308 QString userId = dataParts[1];
309 QString score = dataParts[2];
310 QString coins = dataParts[3];
311 QString level = dataParts[4];
312 QString skin = dataParts[5];
313 QString background = dataParts[6];
314 QString xp = dataParts[7];
315 QString role = dataParts.size() > 8 ? dataParts[8] : "player";
316 QString socketId = dataParts.size() > 9 ? dataParts[9] : "";
317
318 authStatus->setText("Login successful!");
319 authStatus->setStyleSheet("color: green; font-size: 11px;");
320
321 QMessageBox::information(this, "Success", QString("Welcome, %1!\nYour stats:\nScore: %2\nCoins: %3\nLevel: %4")
322 .arg(login).arg(score).arg(coins).arg(level));
323 authBtn->setEnabled(true);
324
325 if (role == "admin")
326 {
327 AdminWindow* adminWindow = new AdminWindow(authLogin->text().trimmed());
328 adminWindow->show();
329
330 QTimer::singleShot(300, this, [this]() {
331 this->close();
332 });
333
334 return;
335 }
336
337 // ЗАПУСК ИГРЫ с передачей всех данных
338 QString pythonPath = "python";
339 // ЗАПУСК ГОТОВОЙ ИГРЫ .EXE
340 QString gamePath =
341 QCoreApplication::applicationDirPath()
342 + "/AlienInvasion/game_launcher.exe";
343
344 QStringList args;
345 args << login;
346 args << "player";
347 args << userId;
348 args << score;
349 args << coins;
350 args << level;
351 args << skin;
352 args << background;
353 args << xp;
354 args << socketId;
355
356 qDebug() << "Starting game:" << gamePath;
357 qDebug() << "Starting game with args:" << args;
358
359 bool started = QProcess::startDetached(gamePath, args);
360
361 if (!started)
362 {
363 QMessageBox::critical(this, "Ошибка", "Не удалось запустить игру.");
364 authBtn->setEnabled(true);
365 return;
366 }
367
368 QTimer::singleShot(300, this, [this]() {
369 this->close();
370 });
371 }
372 }
373 }
374 else if(msg.startsWith("auth_failed"))
375 {
376 QString error = msg.split('&').value(1);
377 authStatus->setText(error);
378 authStatus->setStyleSheet("color: red; font-size: 11px;");
379 authBtn->setEnabled(true);
380 }
381 else if(msg.startsWith("reg_success"))
382 {
383 QString login = msg.split('&').value(1);
384 regStatus->setText("Registration successful!");
385 regStatus->setStyleSheet("color: green; font-size: 11px;");
386 QMessageBox::information(this, "Success", "Registration completed!\nYou can now login.");
387 tabWidget->setCurrentIndex(0);
388 authLogin->setText(login);
389 regBtn->setEnabled(true);
390 }
391 else if(msg.startsWith("reg_failed"))
392 {
393 QString error = msg.split('&').value(1);
394 regStatus->setText(error);
395 regStatus->setStyleSheet("color: red; font-size: 11px;");
396 regBtn->setEnabled(true);
397 }
398 }
