612 static bool buyItem(int userId, int itemId, QString itemType)
613 {
614 if (!db.isOpen()) return false;
615
616 int price = 0;
617 if (itemType == "skin")
618 {
619 if (itemId == 1) price = 100;
620 else if (itemId == 2) price = 250;
621 else return false;
622 }
623 else if (itemType == "bg")
624 {
625 if (itemId == 101) price = 50;
626 else if (itemId == 102) price = 100;
627 else return false;
628 }
629 else return false;
630
631 QSqlQuery check(db);
632 check.prepare("SELECT coins FROM users WHERE id = :id");
633 check.bindValue(":id", userId);
634 if (!check.exec() || !check.next()) return false;
635
636 int coins = check.value(0).toInt();
637 if (coins < price) return false;
638
639 QSqlQuery update(db);
640 update.prepare("UPDATE users SET coins = coins - :price WHERE id = :id");
641 update.bindValue(":price", price);
642 update.bindValue(":id", userId);
643 if (!update.exec()) return false;
644
645 QString table = (itemType == "skin") ? "user_skins" : "user_backgrounds";
646 QString column = (itemType == "skin") ? "skin_id" : "bg_id";
647
648 QSqlQuery insert(db);
649 insert.prepare(QString("INSERT INTO %1 (user_id, %2) VALUES (:user_id, :item_id)")
650 .arg(table).arg(column));
651 insert.bindValue(":user_id", userId);
652 insert.bindValue(":item_id", itemId);
653
654 return insert.exec();
655 }
