88void SingletonClient::slotReadyRead()
89{
90 while (m_pSocket->bytesAvailable() > 0)
91 {
92 QByteArray array = m_pSocket->readAll();
93 m_buffer.append(QString::fromUtf8(array));
94 }
95
96 qDebug() << "Буфер от сервера:" << m_buffer;
97
98 while (m_buffer.contains(QChar(1)))
99 {
100 int separatorIndex = m_buffer.indexOf(QChar(1));
101
102 QString message = m_buffer.left(separatorIndex).trimmed();
103 m_buffer.remove(0, separatorIndex + 1);
104
105 if (!message.isEmpty())
106 emit messageFromServer(message);
107 }
