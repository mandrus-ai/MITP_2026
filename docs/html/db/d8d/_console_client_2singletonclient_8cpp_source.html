122void SingletonClient::slotReadyRead()
123{
124 QString res = "";
125 while (m_pSocket->bytesAvailable() > 0)
126 {
127 QByteArray array = m_pSocket->readAll();
128 res.append(QString::fromUtf8(array));
129 m_buffer.append(QString::fromUtf8(array));
130 }
131
132 qDebug() << "Получено от сервера:" << res;
133
134 if (res.contains('\x01'))
135 {
136 emit messageFromServer(m_buffer);
137 m_buffer.clear();
138 }
139 else
140 {
141 emit messageFromServer(res);
142 }
143}
